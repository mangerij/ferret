QuotientScalarAux

Compute the ratio of two scalar variables.

Example syntax

[AuxScalarKernels<<<{"href": "../../syntax/AuxScalarKernels/index.html"}>>>]
  [./c_saux]
    type = QuotientScalarAux<<<{"description": "Compute the ratio of two scalar variables.", "href": "QuotientScalarAux.html"}>>>
    variable<<<{"description": "The name of the variable that this kernel operates on"}>>> = c
    numerator<<<{"description": "The upstairs of the quotient variable"}>>> = a
    denominator<<<{"description": "The downstairs of the quotient variable"}>>> = b
    execute_on<<<{"description": "The list of flag(s) indicating when this object should be executed. For a description of each flag, see https://mooseframework.inl.gov/source/interfaces/SetupInterface.html."}>>> = 'timestep_end'
  [../]
  [./a_saux]
    type = FunctionScalarAux<<<{"description": "Sets a value of a scalar variable based on a function.", "href": "FunctionScalarAux.html"}>>>
    variable<<<{"description": "The name of the variable that this kernel operates on"}>>> = a
    function<<<{"description": "The functions to set the scalar variable components."}>>> = a_fn
    execute_on<<<{"description": "The list of flag(s) indicating when this object should be executed. For a description of each flag, see https://mooseframework.inl.gov/source/interfaces/SetupInterface.html."}>>> = 'timestep_end'
  [../]
  [./b_saux]
    type = FunctionScalarAux<<<{"description": "Sets a value of a scalar variable based on a function.", "href": "FunctionScalarAux.html"}>>>
    variable<<<{"description": "The name of the variable that this kernel operates on"}>>> = b
    function<<<{"description": "The functions to set the scalar variable components."}>>> = b_fn
    execute_on<<<{"description": "The list of flag(s) indicating when this object should be executed. For a description of each flag, see https://mooseframework.inl.gov/source/interfaces/SetupInterface.html."}>>> = 'timestep_end'
  [../]
[]
(../moose/test/tests/auxscalarkernels/aux_scalar_deps/aux_scalar_deps.i)

Input Parameters

  • variableThe name of the variable that this kernel operates on

    C++ Type:AuxVariableName

    Unit:(no unit assumed)

    Controllable:No

    Description:The name of the variable that this kernel operates on

Required Parameters

  • denominatorThe downstairs of the quotient variable

    C++ Type:std::vector<VariableName>

    Unit:(no unit assumed)

    Controllable:No

    Description:The downstairs of the quotient variable

  • execute_onLINEARThe list of flag(s) indicating when this object should be executed. For a description of each flag, see https://mooseframework.inl.gov/source/interfaces/SetupInterface.html.

    Default:LINEAR

    C++ Type:ExecFlagEnum

    Options:XFEM_MARK, NONE, INITIAL, LINEAR, LINEAR_CONVERGENCE, NONLINEAR, NONLINEAR_CONVERGENCE, POSTCHECK, TIMESTEP_END, TIMESTEP_BEGIN, MULTIAPP_FIXED_POINT_END, MULTIAPP_FIXED_POINT_BEGIN, MULTIAPP_FIXED_POINT_CONVERGENCE, MULTISYSTEM_FIXED_POINT_ITERATION_END, FINAL, CUSTOM

    Controllable:No

    Description:The list of flag(s) indicating when this object should be executed. For a description of each flag, see https://mooseframework.inl.gov/source/interfaces/SetupInterface.html.

  • numeratorThe upstairs of the quotient variable

    C++ Type:std::vector<VariableName>

    Unit:(no unit assumed)

    Controllable:No

    Description:The upstairs of the quotient variable

Optional Parameters

  • control_tagsAdds user-defined labels for accessing object parameters via control logic.

    C++ Type:std::vector<std::string>

    Controllable:No

    Description:Adds user-defined labels for accessing object parameters via control logic.

  • enableTrueSet the enabled status of the MooseObject.

    Default:True

    C++ Type:bool

    Controllable:Yes

    Description:Set the enabled status of the MooseObject.

  • use_displaced_meshFalseWhether or not this object should use the displaced mesh for computation. Note that in the case this is true but no displacements are provided in the Mesh block the undisplaced mesh will still be used.

    Default:False

    C++ Type:bool

    Controllable:No

    Description:Whether or not this object should use the displaced mesh for computation. Note that in the case this is true but no displacements are provided in the Mesh block the undisplaced mesh will still be used.

#
# Testing a solution that is second order in space and first order in time

[Mesh]

  dim = 2
  xmin = 0
  xmax = 1
  ymin = 0

  nx = 10
  ny = 10
  elem_type = QUAD4

[AuxVariables]

    order = FIRST
  [../]
  [./b]
    family = SCALAR
    order = FIRST
  [../]
  [./c]
    family = SCALAR
    order = FIRST
  [../]
[]

[Variables]

    order = FIRST
    family = LAGRANGE
  [../]
[]

[ICs]

    type = ScalarConstantIC
    variable = a
    value = 0
  [../]
  [./b_ic]
    type = ScalarConstantIC
    variable = b
    value = 2
  [../]
[]

[Functions]
  [./exact_fn]
    type = ParsedFunction
    expression = t
  [../]

  [./a_fn]
    type = ParsedFunction
    expression = t
  [../]
  [./b_fn]
    type = ParsedFunction
    expression = (4-t)/2
  [../]
[]

# NOTE: The execute_on = 'timestep_end' is crucial for this test. Without it
# the aux values would be updated during the residual formation and we would
# end up with the right value at the end of the time step. With this flag on,
# the dependencies has to be correct for this test to work. Otherwise the
# values of 'c' will be lagged.

[AuxScalarKernels]
  [./c_saux]
    type = QuotientScalarAux
    variable = c
    numerator = a
    denominator = b
    execute_on = 'timestep_end'
  [../]
  [./a_saux]
    type = FunctionScalarAux
    variable = a
    function = a_fn
    execute_on = 'timestep_end'
  [../]
  [./b_saux]
    type = FunctionScalarAux
    variable = b
    function = b_fn
    execute_on = 'timestep_end'
  [../]
[]

[Kernels]
  [./ie]
    type = TimeDerivative
    variable = u
  [../]

  [./diff]
    type = Diffusion
    variable = u
  [../]
[]

[BCs]
  [./all]
    type = FunctionDirichletBC
    variable = u
    boundary = '0 1 2 3'
    function = exact_fn
  [../]
[]

[Executioner]
  type = Transient
  scheme = 'implicit-euler'

  solve_type = 'PJFNK'
  start_time = 0.0
  num_steps = 2
  dt = 1
[]

[Outputs]
  exodus = true
